import type { Task } from "../task-list/sort";

export interface Session {
  userId: number;
}

const tasks: Task[] = [];
let nextId = 1;

export function resetTasks(): void {
  tasks.length = 0;
  nextId = 1;
}

export function createTask(ownerId: number, title: string, due: string | null): Task {
  const task: Task = { id: nextId++, ownerId, title, due };
  tasks.push(task);
  return task;
}

/** Every read is scoped by owner. */
export function listTasks(ownerId: number): Task[] {
  return tasks.filter((t) => t.ownerId === ownerId);
}

export function findTask(ownerId: number, id: number): Task | undefined {
  return tasks.find((t) => t.ownerId === ownerId && t.id === id);
}

const json = (status: number, body: unknown): Response =>
  new Response(JSON.stringify(body), { status, headers: { "content-type": "application/json" } });

export async function handle(req: Request, session: Session | null): Promise<Response> {
  if (!session) return json(401, { error: "unauthenticated" });
  const url = new URL(req.url);
  if (req.method === "GET" && url.pathname === "/api/tasks") return json(200, listTasks(session.userId));
  return json(404, { error: "not found" });
}
